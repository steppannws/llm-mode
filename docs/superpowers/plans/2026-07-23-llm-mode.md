# llm-mode Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `llm-mode`: a CLI + menu bar app that frees maximum RAM on an M4/24GB Mac, starts an LM Studio server, and exposes it to a client Mac via SSH tunnel — fully reversible.

**Architecture:** Single-file bash CLI (`bin/llm-mode`) holds all logic; a SwiftUI `MenuBarExtra` app and a client tunnel script are thin wrappers around it. State written to `~/.llm-mode/state.json` before any destructive action so `off` can always restore.

**Tech Stack:** bash, bats-core (tests), python3 (JSON helpers), osascript, launchctl, LM Studio `lms` CLI, SwiftUI (macOS 14+).

## Global Constraints

- **Never commit or push** — user's global rule: leave all changes uncommitted. Plan has no commit steps; end each task by checking off its steps.
- All state/config/logs live under `~/.llm-mode/` (override with `LLM_MODE_DIR` env var — tests use this).
- Whitelist (never quit/bootout): `sshd`, `loginwindow`, `WindowServer`, `mDNSResponder`, `notifyd`, `cfprefsd`, `systemstats`, `powerd`, `configd`, `distnoted`, anything matching `LM Studio|lmstudio|lms`, Terminal/iTerm2, and the invoking shell's process chain.
- Default model: `qwen3-coder-30b-a3b` (4-bit MLX). Default port: `1234`. Default wired limit: `20480` MB.
- Every mutating action goes through the `run` wrapper (logs + honors `--dry-run`).
- macOS only; no external deps beyond `bats-core` (dev-only) and LM Studio.

```bash
#!/usr/bin/env bash
# bin/llm-mode — free RAM, run LM Studio server, restore on demand.
set -euo pipefail

STATE_DIR="${LLM_MODE_DIR:-$HOME/.llm-mode}"
STATE_FILE="$STATE_DIR/state.json"
LOG_FILE="$STATE_DIR/log"
DRY_RUN=0
RELAUNCH=0

mkdir -p "$STATE_DIR"

log() { printf '%s %s\n' "$(date '+%Y-%m-%dT%H:%M:%S')" "$*" >>"$LOG_FILE"; }
die() { echo "llm-mode: $*" >&2; exit 1; }
usage() { echo "usage: llm-mode {on|off|status|serve-stop} [--dry-run] [--relaunch]" >&2; exit 2; }

run() {
  log "RUN $*"
  if [[ $DRY_RUN -eq 1 ]]; then echo "[dry-run] $*"; else "$@"; fi
}

cmd_on()        { log "on: start";  run true; log "on: done"; }
cmd_off()       { log "off: start"; run true; log "off: done"; }
cmd_status()    { echo "llm-mode: status not implemented yet"; }
cmd_serve_stop(){ echo "llm-mode: serve-stop not implemented yet"; }

[[ $# -ge 1 ]] || usage
SUBCMD="$1"; shift
for arg in "$@"; do
  case "$arg" in
    --dry-run)  DRY_RUN=1 ;;
    --relaunch) RELAUNCH=1 ;;
    *) usage ;;
  esac
done

case "$SUBCMD" in
  on)         cmd_on ;;
  off)        cmd_off ;;
  status)     cmd_status ;;
  serve-stop) cmd_serve_stop ;;
  *)          usage ;;
esac
```

```bash
state_write() { printf '%s' "$1" >"$STATE_FILE"; log "state written"; }

state_get() {
  python3 -c "
import json, sys
try:
    d = json.load(open('$STATE_FILE'))
    v = d.get('$1', '')
    print('\n'.join(v) if isinstance(v, list) else v)
except Exception:
    pass"

# list visible (non-background) GUI apps, excluding whitelist
list_gui_apps() {
  osascript -e 'tell application "System Events" to get bundle identifier of (processes where background only is false)' \
    | tr ',' '\n' | sed 's/^ *//; s/ *$//' | grep -v '^$' \
    | grep -Ev "$WHITELIST_RE" || true
}

# enabled user launch agents, excluding whitelist
list_agents() {
  launchctl list | awk 'NR>1 {print $3}' \
    | grep -Ev "$WHITELIST_RE" | grep -v '^-$' || true
}

snapshot() {
  local apps agents prev
  apps=$(list_gui_apps | python3 -c 'import json,sys; print(json.dumps([l.strip() for l in sys.stdin if l.strip()]))')
  agents=$(list_agents | python3 -c 'import json,sys; print(json.dumps([l.strip() for l in sys.stdin if l.strip()]))')
  prev=$(sysctl -n iogpu.wired_limit_mb 2>/dev/null || echo 0)
  state_write "{\"apps\": $apps, \"agents\": $agents, \"wired_limit_prev\": \"$prev\", \"forced_kills\": []}"
}
```

```bash
quit_apps() {
  local bid
  while IFS= read -r bid; do
    [[ -n "$bid" ]] || continue
    run osascript -e "tell application id \"$bid\" to quit" || log "quit failed: $bid"
  done < <(state_get apps)
  [[ $DRY_RUN -eq 1 ]] && return 0
  sleep 10
  # force-kill stragglers still running
  while IFS= read -r bid; do
    [[ -n "$bid" ]] || continue
    if pgrep -f "$bid" >/dev/null 2>&1; then
      run pkill -9 -f "$bid" || true
      python3 -c "
import json
d = json.load(open('$STATE_FILE')); d['forced_kills'].append('$bid')
json.dump(d, open('$STATE_FILE','w'))"
      log "forced kill: $bid"
    fi
  done < <(state_get apps)
}

disable_agents() {
  local label found=0
  while IFS= read -r label; do
    [[ -n "$label" ]] || continue
    found=1
    run launchctl bootout "gui/$(id -u)/$label" || log "bootout failed: $label"
    run launchctl disable "gui/$(id -u)/$label" || true
  done < <(state_get agents)
  [[ $found -eq 0 ]] && echo "no agents to disable"
}

pause_services() {
  run sudo mdutil -a -i off
  run sudo tmutil disable
  for label in com.apple.bird com.apple.cloudd com.apple.photoanalysisd; do
    run launchctl bootout "gui/$(id -u)/$label" || true
  done
}

apply_sysctl() {
  run sudo sysctl "iogpu.wired_limit_mb=$CFG_WIRED_MB"
}

cmd_on() {
  log "on: start"
  local before; before=$(vm_stat | awk '/Pages free/ {print $3*16384/1048576 "MB"}')
  snapshot
  quit_apps
  disable_agents
  pause_services
  apply_sysctl
  local after; after=$(vm_stat | awk '/Pages free/ {print $3*16384/1048576 "MB"}')
  echo "free RAM before: $before  after: $after"
  log "on: done"
}
```

```bash
cmd_off() {
  [[ -f "$STATE_FILE" ]] || die "no state file — nothing to restore"
  log "off: start"
  server_stop || true
  run sudo sysctl "iogpu.wired_limit_mb=$(state_get wired_limit_prev)"
  local label
  while IFS= read -r label; do
    [[ -n "$label" ]] || continue
    run launchctl enable "gui/$(id -u)/$label"
    run launchctl bootstrap "gui/$(id -u)" "$HOME/Library/LaunchAgents/$label.plist" 2>/dev/null \
      || run launchctl kickstart "gui/$(id -u)/$label" || log "restore failed: $label"
  done < <(state_get agents)
  run sudo mdutil -a -i on
  run sudo tmutil enable
  if [[ $RELAUNCH -eq 1 ]]; then
    local bid
    while IFS= read -r bid; do
      [[ -n "$bid" ]] || continue
      run open -b "$bid" || true
    done < <(state_get apps)
  fi
  log "off: done"
  echo "restored. reboot fully resets any stubborn daemons."
}
```

```bash
#!/usr/bin/env bash
# client-connect.sh — run on the client Mac; tunnels the LLM API from the server.
set -euo pipefail
HOST="${LLM_HOST:-stepan@m4.local}"
PORT="${LLM_PORT:-1234}"
DRY=0
for a in "$@"; do
  case "$a" in
    --dry-run) DRY=1 ;;
    *@*)       HOST="$a" ;;
    [0-9]*)    PORT="$a" ;;
  esac
done

CMD="ssh -N -L $PORT:localhost:$PORT $HOST"
echo "tunnel: $CMD"
echo "API when up: http://localhost:$PORT/v1  (OpenAI-compatible)"
echo "aider: aider --openai-api-base http://localhost:$PORT/v1 --openai-api-key local"
[[ $DRY -eq 1 ]] && exit 0

$CMD &
TUNNEL_PID=$!
trap 'kill $TUNNEL_PID 2>/dev/null' EXIT
echo "waiting for API..."
for _ in $(seq 1 30); do
  curl -sf -m 2 "http://localhost:$PORT/v1/models" >/dev/null && { echo "API ready."; break; }
  sleep 2
done
wait $TUNNEL_PID
```

- [ ] **Step 2: Status model (shells out to CLI)**

```swift
// app/LLMMode/StatusModel.swift
import Foundation

@MainActor
final class StatusModel: ObservableObject {
    @Published var serverUp = false
    @Published var freeRAM = "…"
    @Published var model = "…"
    @Published var host = "…"
    @Published var busy = false

    private func cli(_ args: [String]) -> String {
        let p = Process()
        p.executableURL = URL(fileURLWithPath: "/usr/local/bin/llm-mode")
        p.arguments = args
        let pipe = Pipe()
        p.standardOutput = pipe
        p.standardError = pipe
        try? p.run()
        p.waitUntilExit()
        let data = pipe.fileHandleForReading.readDataToEndOfFile()
        return String(decoding: data, as: UTF8.self)
    }

    func refresh() {
        let out = cli(["status"])
        for line in out.split(separator: "\n") {
            let parts = line.split(separator: ":", maxSplits: 1).map { $0.trimmingCharacters(in: .whitespaces) }
            guard parts.count == 2 else { continue }
            switch parts[0] {
            case "server":   serverUp = (parts[1] == "up")
            case "free RAM": freeRAM = parts[1]
            case "model":    model = parts[1]
            case "host":     host = parts[1]
            default: break
            }
        }
    }

    func run(_ sub: String) {
        busy = true
        Task.detached { [weak self] in
            _ = await self?.cli([sub])
            await MainActor.run { self?.busy = false; self?.refresh() }
        }
    }
}
```

- [ ] **Step 3: App entry**

```swift
// app/LLMMode/LLMModeApp.swift
import SwiftUI

@main
struct LLMModeApp: App {
    @StateObject private var status = StatusModel()

    var body: some Scene {
        MenuBarExtra("LLM", systemImage: status.serverUp ? "brain.fill" : "brain") {
            Text("\(status.model) — \(status.serverUp ? "up" : "down")")
            Text("free RAM: \(status.freeRAM)")
            Text("host: \(status.host)")
            Divider()
            Button("LLM Mode ON")  { status.run("on") }.disabled(status.busy)
            Button("LLM Mode OFF") { status.run("off") }.disabled(status.busy)
            Button("Refresh")      { status.refresh() }
            Divider()
            Button("Quit") { NSApplication.shared.terminate(nil) }
        }
        .onChange(of: status.busy) { _, _ in status.refresh() }
    }
}
```
